namespace = date_stop

# Daily check. Reschedules itself every day while any alarm is queued.
date_stop.1 = {
	type = country_event
	hidden = yes

	immediate = {
		dsm_migrate_armed = yes
		if = {
			limit = { has_variable = dsm_s1_target }
			dsm_compute_today = yes
			if = {
				limit = { var:dsm_s1_target <= var:dsm_today }
				dsm_fire_due = yes
				# Picked up by the auto-pause widget in gui/timemanip.gui
				set_variable = dsm_pause_pending
				trigger_event = { id = date_stop.2 days = 0 popup = yes }
			}
		}
		if = {
			limit = { has_variable = dsm_s1_target }
			trigger_event = { id = date_stop.1 days = 1 }
		}
		else = {
			remove_variable = dsm_ticking
		}
	}
}

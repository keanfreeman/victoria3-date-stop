namespace = date_stop

# Daily check. Reschedules itself every day while a stop is armed.
date_stop.1 = {
	type = country_event
	hidden = yes

	immediate = {
		if = {
			limit = { NOT = { has_variable = dsm_armed } }
			remove_variable = dsm_ticking
		}
		else = {
			dsm_refresh = yes
			if = {
				limit = { var:dsm_today >= var:dsm_target }
				remove_variable = dsm_armed
				remove_variable = dsm_ticking
				# Picked up by the auto-pause widget in gui/timemanip.gui
				set_variable = dsm_pause_pending
				trigger_event = { id = date_stop.2 days = 0 popup = yes }
			}
			else = {
				trigger_event = { id = date_stop.1 days = 1 }
			}
		}
	}
}

	}

# Date Stop - scripted GUIs. Called from gui/timemanip.gui with the player country as root.

dsm_open_sgui = {
	scope = country
	effect = { dsm_init_target = yes }
}

dsm_arm_sgui = {
	scope = country
	is_valid = { dsm_target_is_future = yes }
	effect = { dsm_arm = yes }
}

dsm_cancel_sgui = {
	scope = country
	is_shown = { has_variable = dsm_armed }
	effect = { dsm_disarm = yes }
}

# is_shown = "the target day arrived and the GUI hasn't paused yet". The effect clears it.
dsm_pause_sgui = {
	scope = country
	is_shown = { has_variable = dsm_pause_pending }
	effect = { remove_variable = dsm_pause_pending }
}

# Used only for showing armed / not armed in the panel.
dsm_armed_sgui = {
	scope = country
	is_shown = { has_variable = dsm_armed }
}

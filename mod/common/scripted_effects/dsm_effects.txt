# Date Stop - scripted effects. All run in the scope of the player's country.
#
# Variables (on the player country):
#   dsm_t_year / dsm_t_month / dsm_t_day   date picked in the panel
#   dsm_target                             picked date as days since 1836.1.1
#   dsm_today_* / dsm_today                today's date (see dsm_generated_effects.txt)
#   dsm_reason / dsm_repeat                picker's reason id and repeat mode (see tools/gen_reasons.py)
#   dsm_s<N>_*                             alarm queue slots, soonest first (see tools/gen_queue.py)
#   dsm_queue_count                        number of queued alarms
#   dsm_f<K>_*                             alarms that fired today, for the popup
#   dsm_ticking                            set while the daily check event is scheduled
#   dsm_pause_pending                      set on an alarm's day; the GUI pauses and clears it
#   dsm_armed*                             the single stop from 1.1.0 and earlier, moved into the queue on load

@dsm_min_year = 1836
@dsm_max_year = 1999

# Target date -> dsm_target (days since 1836.1.1). Also refreshes dsm_t_mlen.
dsm_compute_target = {
	dsm_compute_target_month_info = yes
	set_variable = { name = dsm_target value = var:dsm_t_year }
	change_variable = { name = dsm_target subtract = 1836 }
	change_variable = { name = dsm_target multiply = 365 }
	change_variable = { name = dsm_target add = var:dsm_t_mcum }
	change_variable = { name = dsm_target add = var:dsm_t_day }
	change_variable = { name = dsm_target subtract = 1 }
}

dsm_refresh = {
	dsm_compute_today = yes
	dsm_compute_target = yes
}

# Opening the panel: start the picker at tomorrow the first time, or when
# the picked date has already passed.
dsm_init_target = {
	dsm_migrate_armed = yes
	dsm_init_rel_amount = yes
	if = {
		limit = { NOT = { has_variable = dsm_reason } }
		set_variable = { name = dsm_reason value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = dsm_repeat } }
		set_variable = { name = dsm_repeat value = 0 }
	}
	dsm_compute_today = yes
	if = {
		limit = { has_variable = dsm_t_year }
		dsm_compute_target = yes
		if = {
			limit = { var:dsm_target <= var:dsm_today }
			remove_variable = dsm_t_year
		}
	}
	if = {
		limit = { NOT = { has_variable = dsm_t_year } }
		set_variable = { name = dsm_t_year value = var:dsm_today_year }
		set_variable = { name = dsm_t_month value = var:dsm_today_month }
		set_variable = { name = dsm_t_day value = var:dsm_today_day }
		dsm_change_day = { AMOUNT = 1 }
	}
	dsm_compute_target = yes
}

# Picker buttons can be clicked before the panel's setup has run.
dsm_ensure_picker = {
	if = {
		limit = { NOT = { has_variable = dsm_t_year } }
		dsm_init_target = yes
	}
}

dsm_clamp_year = {
	if = {
		limit = { var:dsm_t_year > @dsm_max_year }
		set_variable = { name = dsm_t_year value = @dsm_max_year }
	}
	if = {
		limit = { var:dsm_t_year < @dsm_min_year }
		set_variable = { name = dsm_t_year value = @dsm_min_year }
	}
}

# Day can't exceed the length of the (possibly new) month.
dsm_clamp_day = {
	dsm_compute_target_month_info = yes
	if = {
		limit = { var:dsm_t_day > var:dsm_t_mlen }
		set_variable = { name = dsm_t_day value = var:dsm_t_mlen }
	}
}

dsm_change_year = {
	dsm_ensure_picker = yes
	change_variable = { name = dsm_t_year add = $AMOUNT$ }
	dsm_clamp_year = yes
	dsm_clamp_day = yes
	dsm_refresh = yes
}





# Add button: queue the picked date with the picker's reason and repeat.
dsm_add = {
	if = {
		limit = {
			dsm_target_is_future = yes
			NOT = { has_variable = dsm_s10_target }
			NOT = { dsm_target_is_queued = yes }
		}
		dsm_refresh = yes
		set_variable = { name = dsm_snew_target value = var:dsm_target }
		set_variable = { name = dsm_snew_year value = var:dsm_t_year }
		set_variable = { name = dsm_snew_month value = var:dsm_t_month }
		set_variable = { name = dsm_snew_day value = var:dsm_t_day }
		# Day of month to return to for monthly/yearly repeats (Jan 31 -> Feb 28 -> Mar 31).
		set_variable = { name = dsm_snew_anchor value = var:dsm_t_day }
		set_variable = { name = dsm_snew_reason value = 0 }
		if = {
			limit = { has_variable = dsm_reason }
			set_variable = { name = dsm_snew_reason value = var:dsm_reason }
		}
		set_variable = { name = dsm_snew_repeat value = 0 }
		if = {
			limit = { has_variable = dsm_repeat }
			set_variable = { name = dsm_snew_repeat value = var:dsm_repeat }
		}
		dsm_queue_add_new = yes
	}
}

# Queues the staged alarm (dsm_snew_*) and makes sure the daily check runs.
dsm_queue_add_new = {
	dsm_new_assign_num = yes
	dsm_queue_insert_new = yes
	dsm_slot_remove = { N = new }
	dsm_queue_refresh = yes
	if = {
		limit = { NOT = { has_variable = dsm_ticking } }
		set_variable = dsm_ticking
		trigger_event = { id = date_stop.1 days = 1 }
	}
}

# 1.1.0 and earlier had a single stop. Move it into the queue.
dsm_migrate_armed = {
	if = {
		limit = { has_variable = dsm_armed }
		if = {
			limit = { has_variable = dsm_armed_target }
			set_variable = { name = dsm_snew_target value = var:dsm_armed_target }
			set_variable = { name = dsm_snew_year value = var:dsm_armed_year }
			set_variable = { name = dsm_snew_month value = var:dsm_armed_month }
			set_variable = { name = dsm_snew_day value = var:dsm_armed_day }
			set_variable = { name = dsm_snew_anchor value = var:dsm_armed_day }
			if = {
				limit = { has_variable = dsm_armed_anchor_day }
				set_variable = { name = dsm_snew_anchor value = var:dsm_armed_anchor_day }
			}
			set_variable = { name = dsm_snew_reason value = 0 }
			if = {
				limit = { has_variable = dsm_armed_reason }
				set_variable = { name = dsm_snew_reason value = var:dsm_armed_reason }
			}
			set_variable = { name = dsm_snew_repeat value = 0 }
			if = {
				limit = { has_variable = dsm_armed_repeat }
				set_variable = { name = dsm_snew_repeat value = var:dsm_armed_repeat }
			}
			dsm_queue_add_new = yes
		}
		remove_variable = dsm_armed
		remove_variable = dsm_armed_target
		remove_variable = dsm_armed_year
		remove_variable = dsm_armed_month
		remove_variable = dsm_armed_day
		remove_variable = dsm_armed_anchor_day
		remove_variable = dsm_armed_reason
		remove_variable = dsm_armed_repeat
		remove_variable = dsm_fired_reason
		remove_variable = dsm_fired_repeat
	}
}

# Fires every alarm that is due (dsm_today must be current). Each one is
# recorded for the popup; repeating ones go back in the queue at their next
# date, the rest are deleted. Sets dsm_f1_reason if anything fired.
dsm_fire_due = {
	dsm_fired_reset = yes
	while = {
		limit = {
			has_variable = dsm_s1_target
			var:dsm_s1_target <= var:dsm_today
		}
		dsm_slot_copy = { FROM = 1 TO = new }
		dsm_queue_delete_1 = yes
		dsm_fired_push = yes
		if = {
			limit = { var:dsm_snew_repeat > 0 }
			dsm_advance_new = yes
			if = {
				limit = { has_variable = dsm_snew_target }
				dsm_fired_set_next = yes
				dsm_queue_insert_new = yes
			}
			else = {
				dsm_fired_set_ended = yes
			}
		}
		dsm_slot_remove = { N = new }
	}
	dsm_queue_refresh = yes
}

# Moves the staged alarm forward by its repeat period until it's after today.
# Removes dsm_snew_target if it can't move (stuck at the last supported year).
# Borrows the picker's date variables for the arithmetic and puts them back afterwards.
dsm_advance_new = {
	if = {
		limit = { has_variable = dsm_t_year }
		set_variable = { name = dsm_saved_t_year value = var:dsm_t_year }
		set_variable = { name = dsm_saved_t_month value = var:dsm_t_month }
		set_variable = { name = dsm_saved_t_day value = var:dsm_t_day }
	}
	set_variable = { name = dsm_advancing value = 1 }

	while = {
		limit = {
			var:dsm_advancing = 1
			var:dsm_snew_target <= var:dsm_today
		}
		set_variable = { name = dsm_t_year value = var:dsm_snew_year }
		set_variable = { name = dsm_t_month value = var:dsm_snew_month }
		set_variable = { name = dsm_t_day value = var:dsm_snew_day }
		if = {
			limit = { var:dsm_snew_repeat = 1 }
			dsm_change_day = { AMOUNT = 7 }
		}
		else_if = {
			limit = { var:dsm_snew_repeat = 2 }
			dsm_change_month = { AMOUNT = 1 }
		}
		else = {
			dsm_change_year = { AMOUNT = 1 }
		}
		if = {
			limit = { var:dsm_snew_repeat >= 2 }
			set_variable = { name = dsm_t_day value = var:dsm_snew_anchor }
			dsm_clamp_day = yes
			dsm_refresh = yes
		}
		if = {
			limit = { var:dsm_target > var:dsm_snew_target }
			set_variable = { name = dsm_snew_target value = var:dsm_target }
			set_variable = { name = dsm_snew_year value = var:dsm_t_year }
			set_variable = { name = dsm_snew_month value = var:dsm_t_month }
			set_variable = { name = dsm_snew_day value = var:dsm_t_day }
		}
		else = {
			set_variable = { name = dsm_advancing value = 0 }
		}
	}
	if = {
		limit = { var:dsm_advancing = 0 }
		remove_variable = dsm_snew_target
	}

	if = {
		limit = { has_variable = dsm_saved_t_year }
		set_variable = { name = dsm_t_year value = var:dsm_saved_t_year }
		set_variable = { name = dsm_t_month value = var:dsm_saved_t_month }
		set_variable = { name = dsm_t_day value = var:dsm_saved_t_day }
		remove_variable = dsm_saved_t_year
		remove_variable = dsm_saved_t_month
		remove_variable = dsm_saved_t_day
		dsm_refresh = yes
	}
	else = {
		# The picker was never opened: leave it unset so it starts at tomorrow.
		remove_variable = dsm_t_year
		remove_variable = dsm_t_month
		remove_variable = dsm_t_day
		dsm_compute_today = yes
	}
	remove_variable = dsm_advancing
}

# ---------------------------------------------------------------------------
# Relative picker: "N days / weeks / months / years from today".
# dsm_rel_amount holds N. The fill effects overwrite the target date fields;
# the player still clicks Add to queue it.
# ---------------------------------------------------------------------------

@dsm_max_rel_amount = 999

dsm_init_rel_amount = {
	if = {
		limit = { NOT = { has_variable = dsm_rel_amount } }
		set_variable = { name = dsm_rel_amount value = 1 }
	}
}

dsm_change_rel_amount = {
	dsm_init_rel_amount = yes
	change_variable = { name = dsm_rel_amount add = $AMOUNT$ }
	if = {
		limit = { var:dsm_rel_amount < 1 }
		set_variable = { name = dsm_rel_amount value = 1 }
	}
	if = {
		limit = { var:dsm_rel_amount > @dsm_max_rel_amount }
		set_variable = { name = dsm_rel_amount value = @dsm_max_rel_amount }
	}
}

# dsm_target (days since 1836.1.1) -> dsm_t_year / dsm_t_month / dsm_t_day
dsm_target_to_ymd = {
	set_variable = { name = dsm_t_year value = @dsm_min_year }
	set_variable = { name = dsm_t_doy value = var:dsm_target }
	while = {
		limit = { var:dsm_t_doy >= 365 }
		change_variable = { name = dsm_t_doy subtract = 365 }
		change_variable = { name = dsm_t_year add = 1 }
	}
	dsm_doy_to_target_md = yes
	remove_variable = dsm_t_doy
}

# Fills the target with today + dsm_rel_amount * $DAYS_PER_UNIT$ days.
dsm_fill_relative_days = {
	dsm_init_rel_amount = yes
	dsm_compute_today = yes
	set_variable = { name = dsm_target value = var:dsm_rel_amount }
	change_variable = { name = dsm_target multiply = $DAYS_PER_UNIT$ }
	change_variable = { name = dsm_target add = var:dsm_today }
	dsm_target_to_ymd = yes
	dsm_clamp_year = yes
	dsm_clamp_day = yes
	dsm_refresh = yes
}

dsm_fill_relative_months = {
	dsm_init_rel_amount = yes
	dsm_compute_today = yes
	set_variable = { name = dsm_t_year value = var:dsm_today_year }
	set_variable = { name = dsm_t_month value = var:dsm_today_month }
	set_variable = { name = dsm_t_day value = var:dsm_today_day }
	change_variable = { name = dsm_t_month add = var:dsm_rel_amount }
	while = {
		limit = { var:dsm_t_month > 12 }
		change_variable = { name = dsm_t_month subtract = 12 }
		change_variable = { name = dsm_t_year add = 1 }
	}
	dsm_clamp_year = yes
	dsm_clamp_day = yes
	dsm_refresh = yes
}

dsm_fill_relative_years = {
	dsm_init_rel_amount = yes
	dsm_compute_today = yes
	set_variable = { name = dsm_t_year value = var:dsm_today_year }
	set_variable = { name = dsm_t_month value = var:dsm_today_month }
	set_variable = { name = dsm_t_day value = var:dsm_today_day }
	change_variable = { name = dsm_t_year add = var:dsm_rel_amount }
	dsm_clamp_year = yes
	dsm_clamp_day = yes
	dsm_refresh = yes
}

# ---------------------------------------------------------------------------
# Step the target by $AMOUNT$ (may be negative). Used by the +/- buttons,
# which send 1 / 5 / 10 for click / shift-click / ctrl-click.
# ---------------------------------------------------------------------------

dsm_change_month = {
	dsm_ensure_picker = yes
	change_variable = { name = dsm_t_month add = $AMOUNT$ }
	while = {
		limit = { var:dsm_t_month > 12 }
		change_variable = { name = dsm_t_month subtract = 12 }
		change_variable = { name = dsm_t_year add = 1 }
	}
	while = {
		limit = { var:dsm_t_month < 1 }
		change_variable = { name = dsm_t_month add = 12 }
		change_variable = { name = dsm_t_year subtract = 1 }
	}
	dsm_clamp_year = yes
	dsm_clamp_day = yes
	dsm_refresh = yes
}

dsm_change_day = {
	dsm_ensure_picker = yes
	dsm_compute_target = yes
	change_variable = { name = dsm_target add = $AMOUNT$ }
	if = {
		limit = { var:dsm_target < 0 }
		set_variable = { name = dsm_target value = 0 }
	}
	dsm_target_to_ymd = yes
	dsm_clamp_year = yes
	dsm_clamp_day = yes
	dsm_refresh = yes
}

# Reason and repeat apply to the next alarm added.
dsm_set_reason = {
	set_variable = { name = dsm_reason value = $ID$ }
}

dsm_set_repeat = {
	set_variable = { name = dsm_repeat value = $ID$ }
}

# Clear all button: deletes every alarm and resets reason and repeat.
dsm_clear = {
	dsm_queue_clear = yes
	dsm_queue_refresh = yes
	set_variable = { name = dsm_reason value = 0 }
	set_variable = { name = dsm_repeat value = 0 }
}
